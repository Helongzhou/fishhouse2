231: /** List-price credits for one generation task from its validated payload; undefined when no price is published. */
232: export function estimateGenerationCredits(model, taskType, params) {
233:     const wire = normalizeModel(model);
234:     const tiers = GENERATION_CREDITS[wire]?.[taskType];

239:         if (params.geometry_quality === 'detailed')
240:             credits += 20;
241:         if (params.quad === true)
242:             credits += 5;
243:         if (params.smart_low_poly === true)
244:             credits += 10;
245:         if (params.generate_parts === true)
246:             credits += 20;
247:     }
248:     return credits;
